package com.example.demo.bean;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;
@Entity
public class Chair {

    @Id
    @Column(name = "c_id")
    private int cId;
    @Column(name = "c_name")
    private String cName;
    @Column(name = "c_legs")
    private int cLegs;
    public Chair() {  // <-- required by JPA/Hibernate
    }
    public Chair(int cId, String cName, int cLegs) {
        this.cId = cId;
        this.cName = cName;
        this.cLegs = cLegs;
    }

    public int getcId() {
        return cId;
    }

    public String getcName() {
        return cName;
    }

    public int getcLegs() {
        return cLegs;
    }

    @Override
    public String toString() {
        return "Chair{" +
                "cId=" + cId +
                ", cName='" + cName + '\'' +
                ", cLegs='" + cLegs + '\'' +
                '}';
    }
}



LoginController.java:


package com.example.demo.controller;

import com.example.demo.bean.Chair;
import com.example.demo.service.LoginService;
import com.example.demo.service.RegisterService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.servlet.ModelAndView;

@Controller("loginController")
public class LoginController {

    @Autowired
    private LoginService ls;
    @Autowired
    RegisterService rservice;

    @GetMapping("/home")
    public String showLoginPage() {
        return "LoginPage";
    }

    @PostMapping("/login")
    public String processLogin(@RequestParam("username") String username,
                               @RequestParam("password") String password,
                               Model model) {

        boolean ok = ls.loginCheck(username, password);

        if (ok) {
            model.addAttribute("user", username);
            return "success";
        } else {
            model.addAttribute("error", "Invalid username or password");
            return "failure";
        }
    }

    @PostMapping("/login-chair")
    public ModelAndView loginChair(@RequestParam int cId, @RequestParam String cName) {
        // Validate chair from DB


        Chair chair = rservice.validateChair(cId, cName);

        if (chair != null) {
            ModelAndView mv = new ModelAndView("success-chair");
            mv.addObject("cId", chair.getcId());
            mv.addObject("cName", chair.getcName());
            mv.addObject("cLegs", chair.getcLegs()); // fetched from DB
            return mv;
        } else {
            return new ModelAndView("failure");
        }

    }

}

import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.servlet.ModelAndView;

@Controller
public class RegisterController {
    @Autowired
    RegisterService rservice;

    @GetMapping("/login-chair-form")
    public String showLoginChairForm() {
        return "LoginChair"; // resolves to /pages/loginchair.jsp
    }

    @GetMapping("/register-form")
    public String showForm() {
        return "register"; // /pages/register.jsp
    }

    @PostMapping("/register-chair")
    public ModelAndView reg(HttpServletRequest request){
        int id=Integer.parseInt(request.getParameter("cId"));
        String name=request.getParameter("cName");
        int legs=Integer.parseInt(request.getParameter("cLegs"));
         Chair e=new Chair(id,name,legs);
        boolean b=rservice.insertDB(e);
        if(b) {
            return new ModelAndView("success");
        }
        else
            return new ModelAndView("failure");
    }
}

package com.example.demo.service;

import com.example.demo.dao.LoginDAO;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

@Service
public class LoginService {

    @Autowired
    private LoginDAO dao;

    /**
     * Delegates to DAO; consistent logic: both username and password must match.
     */
    public boolean loginCheck(String user, String pass) {
        return dao.loginCheck(user, pass);
    }
}

RegisterService.java:

package com.example.demo.service;

import com.example.demo.bean.Chair;
import com.example.demo.dao.RegisterDAO;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

@Service
public class RegisterService {
    @Autowired
    RegisterDAO rdao;
    public boolean insertDB(Chair e){
        rdao.save(e);
        return  true;
    }

    public Chair validateChair(int cId, String cName) {
        return rdao.findByCIdAndCName(cId, cName);
    }

    public boolean validateChairExists(int cId, String cName) {
        return rdao.existsByCIdAndCName(cId, cName); // boolean
    }

    public Chair getChair(int cId, String cName) {
        return rdao.findByCIdAndCName(cId, cName); // entity
    }



}
