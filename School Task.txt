The code is:


def main():
    students = []
    while True:
        try:
            n = int(input("Enter the no of student details to be created: ").strip())
        except:
            print("Invalid Input")
            if students: [print(s) for s in students]
        else:
            if n <= 0:
                print("Invalid Input")
                if students: [print(s) for s in students]
            else:
                for _ in range(n):
                    name = input("Name : ").strip()
                    try:
                        age = int(input("Age : ").strip())
                    except:
                        print("Invalid Input")
                        if students: [print(s) for s in students]
                        continue
                    if age <= 10 or age >= 80:
                        print("Invalid Input")
                        if students: [print(s) for s in students]
                        continue
                    students.append({"Name": name, "Age": age})
        choice = input("Do you want to add more students' details to the list of dictionaries? If yes, press 1, else press 0: ").strip()
        if choice == '1': continue
        print("\nHere's the list of student details :\n")
        [print(s) for s in students]
        print("\nHere's the list of student details sorted with respect to name :")
        [print(s) for s in sorted(students, key=lambda d: d['Name'])]
        if choice != '0': print("Invalid Input")
        break

if __name__ == "__main__":
    main()
